Sub a_ConvertLatex_Final_Pro_Fixed()
    Dim r As Range, fStart As Long, fEnd As Long
    Dim formulaPoints() As Long, count As Long, i As Long
    Dim initialView As Long
    
    ' === БЛОК УСКОРЕНИЯ (оригинальный) ===
    Application.ScreenUpdating = False
    Application.DisplayAlerts = wdAlertsNone
    initialView = ActiveWindow.View.Type
    ActiveWindow.View.Type = wdNormalView ' Быстрый режим
    Options.CheckGrammarAsYouType = False
    Options.CheckSpellingAsYouType = False
    ' =====================================

    ' === БЛОК 1: СТРОЧНЫЕ $$ ... $$ ===
    Set r = ActiveDocument.Range
    count = 0
    With r.Find
        .ClearFormatting
        .text = "\$\$(*)\$\$"
        .MatchWildcards = True
        .Forward = True
        .Wrap = wdFindStop
        Do While .Execute
            count = count + 1
            ReDim Preserve formulaPoints(1 To 2, count)
            formulaPoints(1, count) = r.Start: formulaPoints(2, count) = r.End
            r.Collapse wdCollapseEnd
        Loop
    End With

    If count > 0 Then
        For i = count To 1 Step -1
            fStart = formulaPoints(1, i): fEnd = formulaPoints(2, i)
            Set r = ActiveDocument.Range(fStart, fEnd)
            
            r.text = UniversalTranslit(r.text)
            ActiveDocument.Range(r.End - 1, r.End).Delete
            
            ActiveDocument.Range(fStart + 2, fStart + 2).Select
            
            On Error Resume Next
            Application.Run "MTCommand_TexToggle"
            On Error GoTo 0
            
            Do While ActiveDocument.Range(fStart, fStart + 1).text = "$"
                ActiveDocument.Range(fStart, fStart + 1).Delete
            Loop
        Next i
    End If

    ' === БЛОК 2: ВНУТРИСТРОЧНЫЕ \( ... \) ===
    count = 0: Erase formulaPoints: Set r = ActiveDocument.Range
    With r.Find
        .ClearFormatting
        .text = "\\\((*)\\\)"
        .MatchWildcards = True
        .Forward = True
        .Wrap = wdFindStop
        Do While .Execute
            count = count + 1
            ReDim Preserve formulaPoints(1 To 2, count)
            formulaPoints(1, count) = r.Start: formulaPoints(2, count) = r.End
            r.Collapse wdCollapseEnd
        Loop
    End With

    If count > 0 Then
        For i = count To 1 Step -1
            fStart = formulaPoints(1, i): fEnd = formulaPoints(2, i)
            Set r = ActiveDocument.Range(fStart, fEnd)
            
            ' Добавлена очистка пробелов (Trim), чтобы подхватить "\( V \)"
            Dim s As String
            s = Mid(r.text, 3, Len(r.text) - 4)
            r.text = "$" & UniversalTranslit(Trim(s)) & "$"
            
            ActiveDocument.Range(fStart + 1, fStart + 1).Select
            
            On Error Resume Next
            Application.Run "MTCommand_TexToggle"
            On Error GoTo 0
        Next i
    End If

    ' === ВОЗВРАТ НАСТРОЕК ===
    ActiveWindow.View.Type = initialView
    Options.CheckGrammarAsYouType = True
    Options.CheckSpellingAsYouType = True
    Application.DisplayAlerts = wdAlertsAll
    Application.ScreenUpdating = True
    
    MsgBox "Finish!", vbInformation
End Sub

' Функция транслитерации (Unicode-коды)
Function UniversalTranslit(ByVal txt As String) As String
    Dim i As Long, c As Long, res As String
    res = ""
    For i = 1 To Len(txt)
        c = AscW(Mid(txt, i, 1))
        Select Case c
            Case 1072: res = res & "a": Case 1073: res = res & "b": Case 1074: res = res & "v"
            Case 1075: res = res & "g": Case 1076: res = res & "d": Case 1077: res = res & "e"
            Case 1105: res = res & "yo": Case 1078: res = res & "zh": Case 1079: res = res & "z"
            Case 1080: res = res & "i": Case 1081: res = res & "y": Case 1082: res = res & "k"
            Case 1083: res = res & "l": Case 1084: res = res & "m": Case 1085: res = res & "n"
            Case 1086: res = res & "o": Case 1087: res = res & "p": Case 1088: res = res & "r"
            Case 1089: res = res & "s": Case 1090: res = res & "t": Case 1091: res = res & "u"
            Case 1092: res = res & "f": Case 1093: res = res & "kh": Case 1094: res = res & "ts"
            Case 1095: res = res & "ch": Case 1096: res = res & "sh": Case 1097: res = res & "shch"
            Case 1099: res = res & "y": Case 1101: res = res & "e": Case 1102: res = res & "yu": Case 1103: res = res & "ya"
            Case 1040: res = res & "A": Case 1041: res = res & "B": Case 1042: res = res & "V"
            Case 1043: res = res & "G": Case 1044: res = res & "D": Case 1045: res = res & "E"
            Case 1046: res = res & "Zh": Case 1047: res = res & "Z": Case 1048: res = res & "I"
            Case 1050: res = res & "K": Case 1051: res = res & "L": Case 1052: res = res & "M"
            Case 1053: res = res & "N": Case 1054: res = res & "O": Case 1055: res = res & "P"
            Case 1056: res = res & "R": Case 1057: res = res & "S": Case 1058: res = res & "T"
            Case 1059: res = res & "U": Case 1060: res = res & "F": Case 1061: res = res & "Kh"
            Case 1062: res = res & "Ts": Case 1063: res = res & "Ch": Case 1064: res = res & "Sh"
            Case 1066, 1098, 1068, 1100: res = res & ""
            Case 1069: res = res & "E": Case 1070: res = res & "Yu": Case 1071: res = res & "Ya"
            Case Else: res = res & ChrW(c)
        End Select
    Next i
    UniversalTranslit = res
End Function
